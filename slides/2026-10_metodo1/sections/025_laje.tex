\section{Slab: $g$ ótimo $\times$ verdade}

\begin{frame}{Slab: $g_0 = \tau^*$ em $S$ $\times$ $g^*$ ótimo}
  \footnotesize
  \begin{itemize}\setlength{\itemsep}{0.1em}
    \item Slab $2 \times 1 \times 0{,}15$ (6125 nós), fibra em $x$, $c = (0{,}052;\ 0{,}017)$ cm/ms.
    \item $S$ = paredes $y = 0$ e $x = 0$ (365 nós). Fonte verdadeira: toda a parede $y = 0$ em $t = 0$.
    \item Método 1 com as velocidades verdadeiras, $\varepsilon = 2$ ms. $g_0 = \tau^*$ em $S$;
      $g^*$ = $g$ em $S$ que minimiza o erro de LAT, partindo de $g_0$.
  \end{itemize}
  \medskip
  \centering
  \scriptsize
  \setlength{\tabcolsep}{6pt}
  \begin{tabular}{@{}lcccccc@{}}
    \toprule
    & \shortstack{LAT MAE\\(ms)} & \shortstack{LAT rms\\(ms)} & corr & eik & basins & \shortstack{dispersão\\na fonte (ms)}\\
    \midrule
    verdade & -- & -- & -- & 0,000 & 1 & 0,00\\
    $g_0 = \tau^*$ em $S$ & 0,214 & 0,286 & 0,99987 & 0,005 & 2 & 1,26\\
    $g^*$ (ajuste ao LAT) & 0,017 & 0,028 & 1,00000 & 0,004 & 15 & 0,19\\
    \bottomrule
  \end{tabular}\par
  \medskip
  {\tiny Só LAT. MAE e rms com as médias removidas. eik: RMS de $\|\U\nabla\varphi\| - 1$.
  basins: mínimos locais do mapa. dispersão na fonte: $\max - \min$ de $\varphi$ na parede $y = 0$.}
\end{frame}
