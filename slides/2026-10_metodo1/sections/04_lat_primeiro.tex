\section{Começar do LAT (DTI004)}

\begin{frame}{O teste: um início que já reproduz o LAT}
  \small
  \begin{itemize}\setlength{\itemsep}{0.25em}
    \item Verdade: DTI004, 5 root nodes em $t = 0$ entre os 601 candidatos de Camps; $c = (0{,}065;\ 0{,}050)$ cm/ms; LAT por FIM, ECG por lead field (8 derivações).
    \item $g_0$: o LAT verdadeiro em $S$. $g^*$: o $g$ em $S$ que minimiza o erro de LAT do Método 1.
    \item \textbf{Início A}: a rede ajustada ao LAT.
    \item \textbf{Início B}: a rede ajustada a $g^*$.
    \item Depois, o inverso só com o ECG (1000 iterações, dois tamanhos de passo); velocidades a $\pm 1\%$ da verdade.
  \end{itemize}
\end{frame}
