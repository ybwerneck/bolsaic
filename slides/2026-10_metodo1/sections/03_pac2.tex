\section{O Método 1 nos ventrículos (pac2)}

\begin{frame}{O experimento: a mesma verdade das restrições de fonte}
  \small
  \begin{itemize}\setlength{\itemsep}{0.3em}
    \item Ventrículos pac2 (7387 nós); verdade com duas fontes endocárdicas, VE e VD, em $t = 0$, a 0,86 uma da outra.
    \item Dois desenhos de $S$: \textbf{câmara} ($S$ = lvendo; a fonte do VD fica a 0,50 de $S$) e \textbf{recuperável} ($S$ = os dois endocárdios).
    \item Braços anteriores (Método 0): base de consistência, $c$ na verdade $\pm 5\%$, 20\,000 iterações, 8 sementes (anel recuperável: 3).
    \item Método 1: perda só de ECG, 20\,000 iterações, 3 sementes, $\varepsilon = 2$ ms; $c$ na verdade $\pm 30\%$, partindo de $+10\%$.
    \item Controle: Método 1 com perda de LAT (sem ECG).
  \end{itemize}
  \vfill
  {\scriptsize\color{gray} A comparação não é controlada: caixa de $c$, taxa de aprendizado e penalidade de consistência diferem entre os braços.}
\end{frame}
